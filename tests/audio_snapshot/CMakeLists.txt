cmake_minimum_required(VERSION 3.20)
project(AudioSnapshotComponent LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_path(AUDIO_SNAPSHOT_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
get_filename_component(REPOSITORY "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(audio_snapshot_tests
    "${REPOSITORY}/tests/unit/test_audio_context.cpp"
    "${REPOSITORY}/src/audio_context.cpp")
target_include_directories(audio_snapshot_tests PRIVATE "${REPOSITORY}" "${AUDIO_SNAPSHOT_JSON_INCLUDE_DIR}")
target_link_libraries(audio_snapshot_tests PRIVATE Threads::Threads)
target_compile_definitions(audio_snapshot_tests PRIVATE SUNSHINE_ENABLE_WEBRTC)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(audio_snapshot_tests PRIVATE -Wall -Wextra -Werror)
endif()
option(AUDIO_SNAPSHOT_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
if(AUDIO_SNAPSHOT_SANITIZE)
    target_compile_options(audio_snapshot_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(audio_snapshot_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME audio_snapshot_roles COMMAND audio_snapshot_tests roles.)
add_test(NAME audio_snapshot_ownership COMMAND audio_snapshot_tests context.)
add_test(NAME audio_snapshot_display_transition COMMAND audio_snapshot_tests transition.)
add_test(NAME audio_snapshot_video_owner_unwind COMMAND audio_snapshot_tests --owner-unwind-video)
add_test(NAME audio_snapshot_audio_owner_unwind COMMAND audio_snapshot_tests --owner-unwind-audio)
set_tests_properties(audio_snapshot_roles audio_snapshot_ownership
    audio_snapshot_display_transition
    audio_snapshot_video_owner_unwind audio_snapshot_audio_owner_unwind PROPERTIES TIMEOUT 30)
